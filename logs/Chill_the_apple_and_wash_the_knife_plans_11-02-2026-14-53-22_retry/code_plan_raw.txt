import time
import threading

# Task Description: Chill the apple and wash the knife

# GENERAL TASK DECOMPOSITION
# Independent subtasks:
# SubTask 1: Chill the Apple. (Skills Required: GoToObject, PickupObject, OpenObject, PutObject, CloseObject)
# SubTask 2: Wash the Knife. (Skills Required: GoToObject, PickupObject, PutObject, SwitchOn, SwitchOff)

def chill_apple(robot):
    # 0: SubTask 1: Chill the Apple
    # 1: Go to the Apple.
    GoToObject(robot, 'Apple')
    # 2: Pick up the Apple.
    PickupObject(robot, 'Apple')
    # 3: Go to the Fridge.
    GoToObject(robot, 'Fridge')
    # 4: Open the Fridge.
    OpenObject(robot, 'Fridge')
    # 5: Put Apple in the Fridge
    PutObject(robot, 'Apple', 'Fridge')
    # 6: Close Fridge
    CloseObject(robot, 'Fridge')

def wash_knife(robot):
    # 0: SubTask 2: Wash the Knife
    # 1: Go to the Knife.
    GoToObject(robot, 'Knife')
    # 2: Pick up the Knife.
    PickupObject(robot, 'Knife')
    # 3: Go to the Sink.
    GoToObject(robot, 'Sink')
    # 4: Put the Knife inside the sink
    PutObject(robot, 'Knife', 'Sink')
    # 5: Switch on the Faucet to clean the Knife
    SwitchOn(robot, 'Faucet')
    # 6: Wait for a while to let knife clean.
    time.sleep(5)
    # 7: Switch off Faucet
    SwitchOff(robot, 'Faucet')

# Parallelize SubTask 1 and SubTask 2
robot1 = {'name': 'robot1', 'skills': ['GoToObject', 'OpenObject', 'CloseObject', 'BreakObject', 'SliceObject', 'SwitchOn', 'SwitchOff', 'PickupObject', 'PutObject', 'DropHandObject', 'ThrowObject', 'PushObject', 'PullObject']}
robot2 = {'name': 'robot2', 'skills': ['GoToObject', 'OpenObject', 'CloseObject', 'BreakObject', 'SliceObject', 'SwitchOn', 'SwitchOff', 'PickupObject', 'PutObject','DropHand Object','Throw Object','Push Object','Pull Object']}

task1_thread = threading.Thread(target=chill_apple, args=(robot1,))
task2_thread = threading.Thread(target=wash_knife, args=(robot2,))

# Start executing SubTask 1 and SubTask 2 in parallel
task1_thread.start()
task2_thread.start()

# Wait for both SubTask 1 and SubTask 2 to finish
task1_thread.join()
task2_thread.join()

# Task chill apple and wash knife is done
